const SITE = 'https://tszyeewun.github.io';
const BOT = /bot|crawl|spider|slurp|headless|lighthouse|preview|facebookexternalhit|embedly|monitor|curl|wget|python|java\/|go-http/i;

const SCHEMA = `CREATE TABLE IF NOT EXISTS hits (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  ts INTEGER NOT NULL, path TEXT, title TEXT, ref TEXT,
  country TEXT, region TEXT, city TEXT, postal TEXT, lat REAL, lon REAL, net TEXT,
  device TEXT, browser TEXT, os TEXT, lang TEXT, screen TEXT,
  vid TEXT, sid TEXT, dur INTEGER)`;

let ready = false;
async function init(env) {
  if (ready) return;
  await env.DB.prepare(SCHEMA).run();
  await env.DB.prepare('CREATE INDEX IF NOT EXISTS hits_ts ON hits(ts)').run();
  ready = true;
}

function cors(req) {
  const o = req.headers.get('Origin') || '';
  const ok = o === SITE || o.startsWith('http://localhost') || o.startsWith('http://127.0.0.1');
  return { 'Access-Control-Allow-Origin': ok ? o : SITE, 'Vary': 'Origin' };
}

function parseUA(ua) {
  const device = /iPad|Tablet/i.test(ua) ? 'Tablet' : /Mobi|Android|iPhone/i.test(ua) ? 'Mobile' : 'Desktop';
  const browser = /Edg\//.test(ua) ? 'Edge' : /OPR\//.test(ua) ? 'Opera' : /SamsungBrowser/.test(ua) ? 'Samsung'
    : /CriOS|Chrome\//.test(ua) ? 'Chrome' : /FxiOS|Firefox\//.test(ua) ? 'Firefox' : /Safari\//.test(ua) ? 'Safari' : 'Other';
  const os = /iPhone|iPad|iOS/.test(ua) ? 'iOS' : /Android/.test(ua) ? 'Android' : /Mac OS X/.test(ua) ? 'macOS'
    : /Windows/.test(ua) ? 'Windows' : /Linux/.test(ua) ? 'Linux' : 'Other';
  return { device, browser, os };
}

const cut = (v, n) => (v == null ? null : String(v).slice(0, n));

function authed(req, env) {
  const h = req.headers.get('Authorization') || '';
  return env.DASH_KEY && h === 'Bearer ' + env.DASH_KEY;
}

export default {
  async fetch(req, env) {
    const url = new URL(req.url);
    const p = url.pathname;

    if (req.method === 'OPTIONS') {
      return new Response(null, { headers: { ...cors(req), 'Access-Control-Allow-Methods': 'POST', 'Access-Control-Allow-Headers': 'Content-Type' } });
    }

    if (p === '/hit' && req.method === 'POST') {
      const ua = req.headers.get('User-Agent') || '';
      if (BOT.test(ua)) return new Response('{}', { headers: cors(req) });
      await init(env);
      let b = {};
      try { b = JSON.parse(await req.text()); } catch (e) {}
      const cf = req.cf || {};
      const u = parseUA(ua);
      const r = await env.DB.prepare(
        `INSERT INTO hits (ts,path,title,ref,country,region,city,postal,lat,lon,net,device,browser,os,lang,screen,vid,sid,dur)
         VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,NULL) RETURNING id`
      ).bind(
        Date.now(), cut(b.p, 200), cut(b.t, 200), cut(b.r, 300),
        cf.country || null, cf.region || null, cf.city || null, cf.postalCode || null,
        cf.latitude ? Math.round(cf.latitude * 1000) / 1000 : null,
        cf.longitude ? Math.round(cf.longitude * 1000) / 1000 : null,
        cut(cf.asOrganization, 120),
        u.device, u.browser, u.os, cut(b.l, 20), cut(b.s, 20), cut(b.v, 40), cut(b.ss, 40)
      ).first();
      return new Response(JSON.stringify({ id: r && r.id }), { headers: { ...cors(req), 'Content-Type': 'application/json' } });
    }

    if (p === '/leave' && req.method === 'POST') {
      await init(env);
      let b = {};
      try { b = JSON.parse(await req.text()); } catch (e) {}
      const d = Math.max(0, Math.min(Number(b.d) || 0, 6 * 3600));
      if (b.id) await env.DB.prepare('UPDATE hits SET dur = MAX(COALESCE(dur,0), ?) WHERE id = ?').bind(Math.round(d), Number(b.id)).run();
      return new Response('{}', { headers: cors(req) });
    }

    if (p === '/api/hits') {
      if (!authed(req, env)) return new Response('unauthorised', { status: 401 });
      await init(env);
      const days = Math.min(Number(url.searchParams.get('days')) || 30, 3650);
      const since = Date.now() - days * 86400000;
      const { results } = await env.DB.prepare('SELECT * FROM hits WHERE ts >= ? ORDER BY ts DESC LIMIT 50000').bind(since).all();
      return new Response(JSON.stringify(results), { headers: { 'Content-Type': 'application/json', 'Cache-Control': 'no-store' } });
    }

    if (p === '/' || p === '/dash') {
      return new Response(DASH, { headers: { 'Content-Type': 'text/html; charset=utf-8', 'Cache-Control': 'no-store', 'X-Robots-Tag': 'noindex' } });
    }

    return new Response('not found', { status: 404 });
  }
};
